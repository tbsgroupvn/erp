```typescript
import { Injectable, Logger, OnApplicationShutdown } from '@nestjs/common';
import { ModuleRef, DiscoveryService } from '@nestjs/core';
import { WorkerHost } from '@nestjs/bullmq';

/**
 * Centralized graceful shutdown orchestrator.
 *
 * Shutdown order (enforced by using onApplicationShutdown which fires AFTER onModuleDestroy):
 * 1. Drain BullMQ workers (wait for in-flight jobs, 30s timeout)
 * 2. Close WebSocket server
 * 3. Prisma $disconnect (already handled by PrismaService.onModuleDestroy)
 *
 * Uses onApplicationShutdown (not beforeApplicationShutdown) because:
 * - onModuleDestroy fires first (closes QueueEvents via FailedJobCaptureService)
 * - beforeApplicationShutdown fires next (we drain workers here to ensure DB is still available)
 * - PrismaService.onModuleDestroy closes DB connection (already fired by this point)
 *
 * CORRECTION: We use beforeApplicationShutdown to drain workers BEFORE PrismaService.onModuleDestroy
 * closes the database connection. This ensures in-flight jobs can still access the DB.
 */
@Injectable()
export class GracefulShutdownService implements OnApplicationShutdown {
  private readonly logger = new Logger(GracefulShutdownService.name);
  private readonly DRAIN_TIMEOUT_MS = 30_000;

  constructor(
    private readonly discoveryService: DiscoveryService,
    private readonly moduleRef: ModuleRef,
  ) {}

  async onApplicationShutdown(signal?: string): Promise<void> {
    this.logger.log(`Graceful shutdown initiated (signal: ${signal || 'unknown'})`);
    const start = Date.now();

    // Step 1: Drain BullMQ workers
    await this.drainWorkers();

    // Step 2: Close WebSocket server
    await this.closeWebSocket();

    // Step 3: Prisma disconnect is handled by PrismaService.onModuleDestroy
    // which already fired before onApplicationShutdown

    const elapsed = Date.now() - start;
    this.logger.log(`Graceful shutdown complete in ${elapsed}ms`);
  }

  private async drainWorkers(): Promise<void> {
    // Find all WorkerHost instances via DiscoveryService
    const providers = this.discoveryService.getProviders();
    const workerHosts = providers
      .filter((wrapper) => wrapper.instance instanceof WorkerHost)
      .map((wrapper) => ({
        name: wrapper.instance?.constructor?.name || 'UnknownWorker',
        instance: wrapper.instance as WorkerHost,
      }));

    if (workerHosts.length === 0) {
      this.logger.log('No BullMQ workers to drain');
      return;
    }

    this.logger.log(`Draining ${workerHosts.length} BullMQ worker(s)...`);

    const drainPromise = Promise.allSettled(
      workerHosts.map(async ({ name, instance }) => {
        try {
          await instance.worker.close();
          this.logger.log(`Worker ${name} drained successfully`);
        } catch (err) {
          this.logger.warn(
            `Worker ${name} drain error: ${err instanceof Error ? err.message : String(err)}`,
          );
        }
      }),
    );

    const result = await Promise.race([
      drainPromise.then(() => 'drained' as const),
      new Promise<'timeout'>((resolve) =>
        setTimeout(() => resolve('timeout'), this.DRAIN_TIMEOUT_MS),
      ),
    ]);

    if (result === 'timeout') {
      this.logger.warn(
        `Worker drain timed out after ${this.DRAIN_TIMEOUT_MS}ms — forcing shutdown. ` +
          `Stalled jobs will be retried by BullMQ on next start.`,
      );
    } else {
      this.logger.log('All BullMQ workers drained');
    }
  }

  private async closeWebSocket(): Promise<void> {
    try {
      // Dynamically resolve WsGateway to avoid circular dependency
      // WsGateway may not exist in test environments
      const { WsGateway } = await import('@core/websocket/ws.gateway');
      const wsGateway = this.moduleRef.get(WsGateway, { strict: false });

      if (wsGateway?.server) {
        await new Promise<void>((resolve) => {
          wsGateway.server.close(() => {
            this.logger.log('WebSocket server closed');
            resolve();
          });
          // Safety timeout for WebSocket close
          setTimeout(() => {
            this.logger.warn('WebSocket close timed out after 5s');
            resolve();
          }, 5000);
        });
      }
    } catch (err) {
      // WsGateway may not be available (e.g., in unit tests)
      this.logger.debug(
        `WebSocket close skipped: ${err instanceof Error ? err.message : String(err)}`,
      );
    }
  }
}
```

IMPORTANT: The comment about lifecycle order needs correction. `onApplicationShutdown` fires AFTER `onModuleDestroy`. Since PrismaService.onModuleDestroy closes the DB, workers draining in onApplicationShutdown would fail. Two solutions:

Option A (recommended): Change GracefulShutdownService to implement `BeforeApplicationShutdown` instead, which fires BEFORE `onModuleDestroy`. This way workers drain first, then PrismaService disconnects.

Option B: Move PrismaService to use `onApplicationShutdown` instead of `onModuleDestroy`.

USE OPTION A: Implement `BeforeApplicationShutdown` interface with `beforeApplicationShutdown(signal?: string)` method. This ensures:
1. `beforeApplicationShutdown` fires: drain workers + close WS
2. `onModuleDestroy` fires: PrismaService.$disconnect + FailedJobCaptureService closes QueueEvents
3. `onApplicationShutdown` fires: (nothing needed)

So the final class should be:

    // ... same logic
  }
}
