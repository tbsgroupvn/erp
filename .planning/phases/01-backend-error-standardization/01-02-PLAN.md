import { HttpException, HttpStatus } from '@nestjs/common';
import { ErrorCodeType } from './error-codes';

export class DomainException extends HttpException {
  public readonly errorCode: string;

  constructor(
    errorCode: ErrorCodeType | string,
    message: string,
    status: HttpStatus = HttpStatus.BAD_REQUEST,
  ) {
    super({ message, errorCode }, status);
    this.errorCode = errorCode;
  }
}

**Conversion rules by file:**

**order.service.ts:**
- `throw new Error('Failed to create order...')` -> `throw new DomainException(ErrorCode.ORDER_CREATION_FAILED, 'Failed to create order after 3 attempts', HttpStatus.INTERNAL_SERVER_ERROR)`
- Any other order-related `throw new Error` -> use `ORDER_NOT_FOUND`, `ORDER_INVALID_TRANSITION`, `ORDER_ALREADY_COMPLETED`, `ORDER_ALREADY_CANCELLED`, or `INSUFFICIENT_DEPOSIT` depending on context

**order.repository.ts:**
- Review errors, convert to `DomainException` with appropriate ErrorCode (likely `ORDER_NOT_FOUND` or `INTERNAL_ERROR`)

**order-completion.saga.ts:**
- `throw new Error('Order not found')` -> `throw new DomainException(ErrorCode.ORDER_NOT_FOUND, 'Order not found', HttpStatus.NOT_FOUND)`
- `throw new Error('Already completed')` -> `throw new DomainException(ErrorCode.ORDER_ALREADY_COMPLETED, 'Order already completed', HttpStatus.CONFLICT)`

**auth.service.ts:**
- Auth-related errors -> use `INVALID_CREDENTIALS`, `TOKEN_EXPIRED`, `INVALID_2FA_CODE`, `ACCOUNT_INACTIVE`, `ACCOUNT_LOCKED`, `INVALID_REFRESH_TOKEN` from ErrorCode. Use appropriate HTTP status: 401 for auth failures, 403 for locked/inactive.

**api-key-rotation.service.ts:**
- `throw new Error('API key not found')` -> `throw new DomainException(ErrorCode.API_KEY_NOT_FOUND, 'API key not found', HttpStatus.NOT_FOUND)`

**prisma.service.ts:**
- Transaction error -> `throw new DomainException(ErrorCode.TRANSACTION_ERROR, message, HttpStatus.INTERNAL_SERVER_ERROR)`

**query-analyzer.service.ts:**
- Validation errors -> `throw new BadRequestException(message)` (NestJS built-in, will get VALIDATION_ERROR via deriveErrorCode)

**vault.service.ts:**
- Vault errors -> `throw new DomainException(ErrorCode.VAULT_ERROR, message, HttpStatus.INTERNAL_SERVER_ERROR)`

**encryption.service.ts:**
- Crypto errors -> `throw new DomainException(ErrorCode.ENCRYPTION_ERROR, message, HttpStatus.INTERNAL_SERVER_ERROR)`

**key-rotation.service.ts:**
- Key rotation errors -> `throw new DomainException(ErrorCode.ENCRYPTION_ERROR, message, HttpStatus.INTERNAL_SERVER_ERROR)`

**business-hours.ts:**
- `throw new Error('workEnd must be greater than workStart')` -> `throw new BadRequestException('workEnd must be greater than workStart')` (import from `@nestjs/common`)
