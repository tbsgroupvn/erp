```typescript
import { randomUUID } from 'crypto';

export interface WsErrorResponse {
  errorCode: string;
  message: string;
  requestId: string;
  timestamp: string;
}

export function wsError(
  errorCode: string,
  message: string,
  requestId?: string,
): WsErrorResponse {
  return {
    errorCode,
    message,
    requestId: requestId || 'ws-' + randomUUID().slice(0, 8),
    timestamp: new Date().toISOString(),
  };
}
```

**3. Create `src/core/events/processors/processor-error.util.ts`:**

```typescript
import { Logger } from '@nestjs/common';
import { Job } from 'bullmq';
import { DomainException } from '@common/exceptions';
import { ErrorCode } from '@common/exceptions';

export interface ProcessorErrorLog {
  errorCode: string;
  message: string;
  requestId: string;
  jobId: string;
  queue: string;
  eventType: string;
  attemptsMade: number;
}

export function logProcessorError(
  logger: Logger,
  job: Job,
  error: Error | unknown,
): void {
  const errorObj = error instanceof Error ? error : new Error(String(error));
  const correlationId = job.data?.metadata?.correlationId || job.id || 'unknown';

  const logEntry: ProcessorErrorLog = {
    errorCode: error instanceof DomainException ? error.errorCode : ErrorCode.JOB_PROCESSING_FAILED,
    message: errorObj.message,
    requestId: correlationId,
    jobId: job.id || 'unknown',
    queue: job.queueName,
    eventType: job.data?.type || job.name || 'unknown',
    attemptsMade: job.attemptsMade,
  };

  logger.error(JSON.stringify(logEntry), errorObj.stack);
}

export function buildProcessorErrorLog(
  job: Job,
  error: Error | unknown,
): ProcessorErrorLog {
  const errorObj = error instanceof Error ? error : new Error(String(error));
  const correlationId = job.data?.metadata?.correlationId || job.id || 'unknown';

  return {
    errorCode: error instanceof DomainException ? error.errorCode : ErrorCode.JOB_PROCESSING_FAILED,
    message: errorObj.message,
    requestId: correlationId,
    jobId: job.id || 'unknown',
    queue: job.queueName,
    eventType: job.data?.type || job.name || 'unknown',
    attemptsMade: job.attemptsMade,
  };
}
```
